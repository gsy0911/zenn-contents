# コード

コンストラクタを以下のように作成します。

```typescript: ./constructs/ses.ts
import { Duration, aws_ses } from "aws-cdk-lib";
import { Construct } from "constructs";

import * as route53 from "aws-cdk-lib/aws-route53";

interface ISes {
  domainName: string;
}

export class Ses extends Construct {
  constructor(scope: Construct, id: string, props: ISes) {
    super(scope, id);

    const { domainName } = props;
    const region = "ap-northeast-1";
    const mailFromDomain = `mail.${domainName}`;

    // 既存のホストゾーンを取得
    const hostedZone = route53.PublicHostedZone.fromLookup(this, "HostedZone", {
      domainName,
    });

    // SES Domain Identity
    // MEMO: DKIM レコードの追加は既定で自動的に行われる
    new aws_ses.EmailIdentity(this, "SesEmailIdentity", {
      identity: aws_ses.Identity.publicHostedZone(hostedZone),
      // DMARC対応
      mailFromDomain,
      mailFromBehaviorOnMxFailure: aws_ses.MailFromBehaviorOnMxFailure.USE_DEFAULT_VALUE,
    });

    // MXレコードの追加
    new route53.MxRecord(this, "SESMXRecord", {
      zone: hostedZone,
      values: [
        {
          hostName: `inbound-smtp.${region}.amazonaws.com`,
          priority: 10,
        },
      ],
      ttl: Duration.minutes(5),
    });

    // DMARC対応
    new route53.TxtRecord(this, "SesTxtRecord", {
      zone: hostedZone,
      recordName: `_dmarc.${domainName}`,
      values: ["v=DMARC1; p=none;"],
      ttl: Duration.hours(1),
    });
  }
}
```

```typescript: ./lib/stack.ts
import * as cdk from "aws-cdk-lib/core";
import { Construct } from "constructs";
// import * as sqs from 'aws-cdk-lib/aws-sqs';
import { Ses } from "../constructs/ses";

export class Apne1PapersMiddleware extends cdk.Stack {
  constructor(scope: Construct, id: string, props?: cdk.StackProps) {
    super(scope, id, props);

    new Ses(this, "Ses", { domainName: "acxl.site" });
  }
}
```

## 動作確認

上記のコンストラクをデプロイしてboto3を使って送信確認をしてみます。
次のスクリプトを実行すると初期設定で追加メールアドレスに送信されています。

```python
import boto3

ses = boto3.client(
    "ses",
    region_name="ap-northeast-1",
)
response = ses.send_email(
    Source="Sender <no-reply@{your-domain}>",
    Destination={
        "ToAddresses": ["{最初に設定したメールアドレス}"],
    },
    Message={
        "Subject": {
            "Data": "テストメール",
            "Charset": "UTF-8",
        },
        "Body": {
            "Text": {
                "Data": "これはSESから送信したテストメールです。",
                "Charset": "UTF-8",
            }
        },
    },
)
```
